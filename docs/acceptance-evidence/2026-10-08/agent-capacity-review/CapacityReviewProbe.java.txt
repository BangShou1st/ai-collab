package com.shitulelv.aicollab.agent.application.runtime;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.shitulelv.aicollab.agent.application.view.AgentRunView;
import com.shitulelv.aicollab.agent.application.view.AgentStepView;
import com.shitulelv.aicollab.agent.domain.model.*;
import com.shitulelv.aicollab.agent.infrastructure.repository.AgentRepository;
import com.shitulelv.aicollab.infrastructure.ai.turn.*;
import org.mockito.ArgumentCaptor;

import java.time.OffsetDateTime;
import java.util.*;

import static org.mockito.ArgumentMatchers.*;
import static org.mockito.Mockito.*;

public class CapacityReviewProbe {
    private static final ObjectMapper JSON = new ObjectMapper().findAndRegisterModules();
    private static int failures;

    public static void main(String[] args) {
        probe("committedSummaryReachesComposer", CapacityReviewProbe::committedSummaryReachesComposer);
        probe("partialSourceDoesNotAdvanceWholeStep", CapacityReviewProbe::partialSourceDoesNotAdvanceWholeStep);
        probe("largeWindowPreservesOlderRawEvidence", CapacityReviewProbe::largeWindowPreservesOlderRawEvidence);
        System.out.println("Expected-correctness assertions failed: " + failures + "/3");
        if (failures != 0) System.exit(1);
    }

    private static void probe(String name, Runnable task) {
        try {
            task.run();
            System.out.println("PASS " + name);
        } catch (AssertionError failure) {
            failures++;
            System.out.println("FAIL " + name + ": " + failure.getMessage());
        }
    }

    private static AgentRunView run() {
        OffsetDateTime now = OffsetDateTime.now();
        return new AgentRunView(UUID.randomUUID(), UUID.randomUUID(), UUID.randomUUID(), UUID.randomUUID(),
                null, "SUPERVISOR", 0, "Research document evidence", AgentRunStatus.RUNNING,
                64, 64, 3, null, null, 0, 0, 0, 0, 0, 0, 0, false,
                false, false, 0, null, null, null, "PROJECT_RESEARCH", 1, 2, false, now, now);
    }

    private static AgentRepository repository() {
        AgentRepository repository = mock(AgentRepository.class);
        when(repository.listRecentMessages(any(), anyInt())).thenReturn(new ArrayList<>());
        when(repository.citationsStillValid(any(), any())).thenReturn(true);
        return repository;
    }

    private static AgentExecutionContext context(AgentRunView run) {
        return new AgentExecutionContext(run.projectId(), run.sessionId(), run.id(), run.requesterId(),
                "SUPERVISOR", false, AgentPageContext.empty(), AgentResourcePolicy.v2Limits(0), 0, List.of());
    }

    private static AgentSkill skill() {
        AgentSkill skill = mock(AgentSkill.class);
        when(skill.code()).thenReturn("PROJECT_RESEARCH");
        when(skill.instruction()).thenReturn("Read only document research");
        when(skill.outputContract()).thenReturn("Evidence and gaps");
        return skill;
    }

    private static AgentStepView step(int sequence, String body) {
        return new AgentStepView(UUID.randomUUID(), sequence, AgentStepType.TOOL_CALL_COMPLETED,
                "read_document_section", JSON.createObjectNode().put("toolCallId", "call-" + sequence),
                JSON.createObjectNode().put("status", "SUCCEEDED").put("content", body),
                "TOOL_SUCCESS", null, null, false, null, null, OffsetDateTime.now());
    }

    private static String messages(AgentModelMessageComposer.Composition composition) {
        return composition.messages().toString();
    }

    private static void committedSummaryReachesComposer() {
        AgentRunView run = run();
        AgentRepository repository = repository();
        when(repository.latestRunContextSummary(run.projectId(), run.id())).thenReturn(
                JSON.createObjectNode().put("scope", "RUN_CONTEXT").put("status", "COMMITTED")
                        .put("sourceThroughSequence", 5).put("text", "SUMMARY_ONLY_FACT_4907"));
        AgentModelMessageComposer composer = new AgentModelMessageComposer(repository, null, JSON);
        var composition = composer.composeV2(run, context(run), skill(),
                AgentPlan.create(run.goal(), List.of()), List.of(), 950_000, 1.0, false);
        if (!messages(composition).contains("SUMMARY_ONLY_FACT_4907")) {
            throw new AssertionError("committed RUN_CONTEXT text absent; repository read count="
                    + mockingDetails(repository).getInvocations().stream()
                    .filter(i -> i.getMethod().getName().equals("latestRunContextSummary")).count());
        }
    }

    private static void partialSourceDoesNotAdvanceWholeStep() {
        AgentRunView run = run();
        AgentRepository repository = repository();
        RoutingAgentModelExecutor model = mock(RoutingAgentModelExecutor.class);
        AgentStepView huge = step(1, "BEGIN_MARKER" + "x".repeat(80_000) + "UNSENT_TAIL_7193");
        AgentStepView latest = new AgentStepView(UUID.randomUUID(), 2, AgentStepType.MODEL_REQUEST,
                null, null, null, "latest", null, null, false, null, null, OffsetDateTime.now());
        when(repository.listSteps(run.projectId(), run.id())).thenReturn(List.of(huge, latest));
        when(repository.beginRunContextAttempt(any(), anyInt(), anyInt(), anyInt()))
                .thenReturn(UUID.randomUUID());
        when(model.callModelWithoutTools(any(), any())).thenReturn(new ModelTurnResult(
                "Controlled summary of supplied prefix", List.of(), ModelFinishReason.STOP,
                new ModelUsage(20_000, 15), "controlled", "controlled", 1L));
        var summarizer = new AgentContextSummarizer(repository, model, JSON);
        var budget = new AgentContextBudget.Budget(950_000, 950_000, 256_000, 128_000,
                AgentContextBudget.BINDING_MODEL_WINDOW, false);
        boolean committed = summarizer.compactRunContext(run, null, budget, Integer.MAX_VALUE, () -> true);
        ArgumentCaptor<List<ModelMessage>> request = ArgumentCaptor.forClass(List.class);
        verify(model).callModelWithoutTools(eq(run), request.capture());
        ArgumentCaptor<JsonNode> summary = ArgumentCaptor.forClass(JsonNode.class);
        verify(repository).completeRunContextAttempt(any(), eq("COMMITTED"), anyString(), any(), any(), summary.capture());
        boolean tailVisible = request.getValue().toString().contains("UNSENT_TAIL_7193");
        if (committed && !tailVisible && summary.getValue().path("sourceThroughSequence").asInt() >= 1) {
            throw new AssertionError("unsent tail nevertheless covered: sourceThroughSequence="
                    + summary.getValue().path("sourceThroughSequence") + ", sourceTruncated="
                    + summary.getValue().path("sourceTruncated"));
        }
    }

    private static void largeWindowPreservesOlderRawEvidence() {
        AgentRunView run = run();
        AgentRepository repository = repository();
        AgentStepView older = step(1, "OLD_PREFIX" + "x".repeat(24_000) + "OLDER_REQUIRED_FACT_8711");
        AgentStepView latest = step(2, "Recent independent observation");
        AgentModelMessageComposer composer = new AgentModelMessageComposer(repository, null, JSON);
        var composition = composer.composeV2(run, context(run), skill(), AgentPlan.create(run.goal(), List.of()),
                List.of(older, latest), 950_000, 1.0, false);
        if (!messages(composition).contains("OLDER_REQUIRED_FACT_8711")) {
            throw new AssertionError("older source projected before pressure: availableInput=950000, charsUsed="
                    + composition.stats().charsUsed() + ", projectedOutputs=" + composition.stats().toolResultsProjected());
        }
    }
}
