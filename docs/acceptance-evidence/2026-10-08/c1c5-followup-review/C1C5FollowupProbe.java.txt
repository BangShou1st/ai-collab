    private static String status(UUID attempt) {
        return AgentRunContextCommitPostgresTest.jdbc.queryForObject("SELECT output_json->>'status' FROM agent_step WHERE id=?", String.class, attempt);
    }
    private static void expiredClaimCannotPublishRunSummary() {
        Fixture f = fixture();
        JdbcTemplate jdbc = AgentRunContextCommitPostgresTest.jdbc;
        jdbc.update("UPDATE agent_run SET lease_expires_at=now()-interval '1 second' WHERE id=?", f.run().id());
        var tx = new TransactionTemplate(new DataSourceTransactionManager(jdbc.getDataSource()));
        try (var scope = new AgentLeaseScope(1)) {
            tx.executeWithoutResult(s -> complete(AgentRunContextCommitPostgresTest.recorder, f.attempt()));
        }
        if ("COMMITTED".equals(status(f.attempt()))) {
            throw new AssertionError("expired claim epoch=1 published COMMITTED; actual input="
                    + jdbc.queryForObject("SELECT input_tokens_actual FROM agent_run WHERE id=?", Long.class, f.run().id()));
        }
    }
    private static final class GateJdbc extends JdbcTemplate {
        final CountDownLatch checked = new CountDownLatch(1), publish = new CountDownLatch(1);
        GateJdbc(DataSource ds) { super(ds); }
        @Override public List<Map<String,Object>> queryForList(String sql, Object... args) {
            var result = super.queryForList(sql, args);
            if (sql.contains("JOIN agent_run r") && sql.contains("goal_revision")) {
                checked.countDown();
                try { if (!publish.await(5, TimeUnit.SECONDS)) throw new AssertionError("publish gate timed out"); }
                catch (InterruptedException e) { Thread.currentThread().interrupt(); throw new RuntimeException(e); }
            }
            return result;
        }
    }
    private static void cancelBetweenCheckAndPublishCannotPublish() throws Exception {
        Fixture f = fixture();
        JdbcTemplate jdbc = AgentRunContextCommitPostgresTest.jdbc;
        var gate = new GateJdbc(jdbc.getDataSource());
        var recorder = new AgentRunEventRecorder(gate, JSON);
        var tx = new TransactionTemplate(new DataSourceTransactionManager(jdbc.getDataSource()));
        var threads = Executors.newFixedThreadPool(2);
        try {
            Future<?> publication = threads.submit(() -> {
                try (var scope = new AgentLeaseScope(1)) {
                    tx.executeWithoutResult(s -> complete(recorder, f.attempt()));
                }
            });
            if (!gate.checked.await(5, TimeUnit.SECONDS)) throw new AssertionError("fence read was not reached");
            Future<?> cancellation = threads.submit(() -> jdbc.update("UPDATE agent_run SET cancel_requested_at=now() WHERE id=?", f.run().id()));
            boolean cancellationCommittedFirst = true;
            try { cancellation.get(2, TimeUnit.SECONDS); }
            catch (TimeoutException blockedByCorrectLock) { cancellationCommittedFirst = false; }
            gate.publish.countDown();
            publication.get(5, TimeUnit.SECONDS);
            cancellation.get(5, TimeUnit.SECONDS);
            if (cancellationCommittedFirst && "COMMITTED".equals(status(f.attempt()))) {
                throw new AssertionError("cancel committed after fence read, before publication; summary still COMMITTED");
            }
        } finally {
            gate.publish.countDown();
            threads.shutdownNow();
            threads.awaitTermination(5, TimeUnit.SECONDS);
        }
    }
