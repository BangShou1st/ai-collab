import static org.mockito.Mockito.*;

/** Temporary review evidence only; archive outside the formal test tree after execution. */
class ScratchStreamingReviewProbeTest {
    private final ObjectMapper json = new ObjectMapper();

    @Test
    @SuppressWarnings("unchecked")
    void slowPresentationMustNotTurnAnImmediateProviderResponseIntoModelTimeout() throws Exception {
        UUID projectId = UUID.randomUUID();
        UUID runId = UUID.randomUUID();
        AgentEventStreamService streams = new AgentEventStreamService(
                mock(ProjectAccessGuard.class), mock(AgentRepository.class), mock(AgentEventRepository.class));
        SseEmitter emitter = mock(SseEmitter.class);
        CountDownLatch enteredSend = new CountDownLatch(1);
        CountDownLatch releaseSend = new CountDownLatch(1);
        doAnswer(invocation -> {
            enteredSend.countDown();
            releaseSend.await(10, TimeUnit.SECONDS);
            return null;
        }).when(emitter).send(any(SseEmitter.SseEventBuilder.class));
        Class<?> subscriptionClass = Class.forName(AgentEventStreamService.class.getName() + "$Subscription");
        Constructor<?> constructor = subscriptionClass.getDeclaredConstructor(UUID.class, UUID.class, SseEmitter.class, long.class);
        constructor.setAccessible(true);
        Object subscription = constructor.newInstance(projectId, runId, emitter, 0L);
        Field subscriptionsField = AgentEventStreamService.class.getDeclaredField("subscriptions");
        subscriptionsField.setAccessible(true);
        Map<UUID, CopyOnWriteArrayList<Object>> subscriptions =
                (Map<UUID, CopyOnWriteArrayList<Object>>) subscriptionsField.get(streams);
        subscriptions.put(runId, new CopyOnWriteArrayList<>(List.of(subscription)));

        HttpClient transport = mock(HttpClient.class);
        HttpResponse<InputStream> response = mock(HttpResponse.class);
        when(response.statusCode()).thenReturn(200);
        when(response.headers()).thenReturn(HttpHeaders.of(Map.of("Content-Type", List.of("text/event-stream")), (a, b) -> true));
        String immediateWire = "data: {\"choices\":[{\"delta\":{\"content\":\"provider replied immediately\"},\"finish_reason\":\"stop\"}]}\n\ndata: [DONE]\n\n";
        when(response.body()).thenAnswer(invocation -> new ByteArrayInputStream(immediateWire.getBytes(StandardCharsets.UTF_8)));
        when(transport.send(any(HttpRequest.class), org.mockito.ArgumentMatchers.<HttpResponse.BodyHandler<InputStream>>any()))
                .thenReturn(response);
        JsonHttpModelClient http = new JsonHttpModelClient(json, mock(OutboundEndpointPolicy.class), transport, Duration.ofSeconds(1));
        OpenAiCompatibleModelAdapter adapter = new OpenAiCompatibleModelAdapter(json, http);
        ModelConfiguration config = new ModelConfiguration(UUID.randomUUID(), projectId, "fixture", ModelProviderType.OPENAI_COMPATIBLE,
                "https://example.com", "/v1/chat/completions", null, "fixture-model", true, 0.2, 1200,
                EnumSet.of(ModelCapability.CHAT, ModelCapability.STREAMING, ModelCapability.NATIVE_TOOLS), OffsetDateTime.now(), OffsetDateTime.now());
        ModelTurnCommand command = new ModelTurnCommand(ModelPurpose.AGENT, List.of(new ModelMessage.User("fixture")), List.of(), false);
        AgentEventService events = new AgentEventService(mock(AgentEventRepository.class), streams);
        BusinessException observedFailure = null;
        try {
            // Identical wire data completes successfully before adding the presentation subscriber.
            assertThat(adapter.turnWithSession(config, "fixture-key", command, AiRequestMetadata.fresh(), "fixture").content())
                    .isEqualTo("provider replied immediately");
            ModelContentPreview.activate(new AgentContentPreviewPublisher(events, json, projectId, runId, "fixture-call"));
            try {
                adapter.turnWithSession(config, "fixture-key", command, AiRequestMetadata.fresh(), "fixture");
            } catch (BusinessException failure) {
                observedFailure = failure;
            }
        } finally {
            ModelContentPreview.clear();
            releaseSend.countDown();
            http.close();
        }
        assertThat(enteredSend.getCount()).isZero();
        System.out.println("STREAM_REVIEW_SLOW_SUBSCRIBER=" + (observedFailure == null ? "SUCCESS" : observedFailure.getErrorCode()));
        assertThat(observedFailure).as("An optional presentation subscriber must not turn a ready model response into a timeout").isNull();
    }
}
